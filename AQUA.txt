    const KZ_BOUNDS = L.latLngBounds([40.4, 46.3], [55.6, 87.5]);

    // Initialize Map
    function initMap() {
      const mapContainer = document.getElementById('kz-map');
      if (!mapContainer) return;

      map = L.map('kz-map', {
        center: [48.1, 67.5],
        zoom: 5,
        minZoom: 4,
        maxZoom: 12,
        maxBounds: KZ_BOUNDS.pad(0.18),
        maxBoundsViscosity: 0.85
      });

      L.tileLayer('https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png', {
        attribution: '&copy; OpenStreetMap &copy; CARTO | AQUA KZ',
        subdomains: 'abcd',
        maxZoom: 18
      }).addTo(map);

      L.geoJSON(KAZAKHSTAN_BORDER_GEOJSON, {
        style: {
          color: '#0284C7',
          weight: 2.2,
          opacity: 0.75,
          fillColor: '#38BDF8',
          fillOpacity: 0.06,
          dashArray: '4, 4'
        },
        interactive: false
      }).addTo(map);

      cityMarkersLayer.addTo(map);
      hydroStationsLayer.addTo(map);

      // Populate Quick Select Dropdown
      const select = document.getElementById('city-quick-select');
      select.innerHTML = '<option value="">Қаланы таңдаңыз (20 қала)...</option>';
      CITIES_DATA.forEach(city => {
        const opt = document.createElement('option');
        opt.value = city.id;
        opt.textContent = `${city.name} (${city.region.split(' ')[0]})`;
        select.appendChild(opt);
      });

      map.on('zoomend', () => {
        const z = map.getZoom();
        const hintEl = document.getElementById('zoom-status-text');
        if (z >= 6) {
          renderHydroStations();
          hintEl.textContent = `Zoom ${z}: Өзен-көл гидробекеттері көрсетілуде`;
        } else {
          hydroStationsLayer.clearLayers();
          hintEl.textContent = 'Гидробекеттер үшін картаны үлкейтіңіз (Zoom ≥ 6)';
        }
      });

      renderMapMarkers();
      updateLegend();
      renderCityPanel(CITIES_DATA[0]);

      setTimeout(() => {
        if (map) {
          map.invalidateSize();
          map.fitBounds(KZ_BOUNDS, { padding: [20, 20] });
        }
      }, 250);
    }

    function resetKazakhstanMapView() {
      if (!map) return;
      map.invalidateSize();
      map.flyToBounds(KZ_BOUNDS, { padding: [20, 20], duration: 0.7 });
    }

    function renderMapMarkers() {
      cityMarkersLayer.clearLayers();
      const cfg = LAYER_CONFIGS[currentLayer];

      CITIES_DATA.forEach(city => {
        const color = cfg.getColor(city);
        const valText = cfg.getBadgeValue(city);
        const isSelected = city.id === selectedCityId;

        const icon = L.divIcon({
          className: '',
          html: `<div class="iqair-marker ${isSelected ? 'iqair-marker-active' : ''}" style="background-color: ${color}; width: 38px; height: 38px;" title="${city.name}">${valText}</div>`,
          iconSize: [38, 38],
          iconAnchor: [19, 19]
        });

        const marker = L.marker([city.lat, city.lng], { icon });
        marker.bindTooltip(city.name, {
          permanent: true,
          direction: 'bottom',
          offset: [0, 14],
          className: 'city-map-label'
        });

        marker.on('click', () => {
          selectedCityId = city.id;
          renderMapMarkers();
          renderCityPanel(city);
          document.getElementById('city-quick-select').value = city.id;
        });

        cityMarkersLayer.addLayer(marker);
      });
    }

    function renderHydroStations() {
      hydroStationsLayer.clearLayers();
      HYDRO_STATIONS.forEach(st => {
        const icon = L.divIcon({
          className: '',
          html: `<div class="hydro-station-marker px-2 py-1" title="${st.name}">💧 ${st.river.split(' ')[0]}</div>`,
          iconSize: [94, 24],
          iconAnchor: [47, 12]
        });

        const m = L.marker([st.lat, st.lng], { icon });
        m.bindPopup(`
          <div class="text-xs font-sans p-1">
            <p class="font-bold text-[#0A2540]">${st.name}</p>
            <p class="text-slate-600 mt-1">Су деңгейі: <strong>${st.level}</strong></p>
            <p class="text-slate-600">Температура: <strong>${st.temp}</strong></p>
            <p class="text-slate-600">Су сапасы: <strong>${st.quality}</strong></p>
            <p class="text-[10px] text-sky-600 mt-1">Дереккөз: Қазгидромет</p>
          </div>
        `);
        hydroStationsLayer.addLayer(m);
      });
    }

    function switchMapLayer(layerKey) {
      currentLayer = layerKey;
      document.querySelectorAll('.layer-btn').forEach(btn => {
        if (btn.dataset.layer === layerKey) {
          btn.classList.remove('bg-white', 'text-slate-700', 'border', 'border-slate-200');
          btn.classList.add('bg-[#0A2540]', 'text-white', 'shadow-sm');
        } else {
          btn.classList.add('bg-white', 'text-slate-700', 'border', 'border-slate-200');
          btn.classList.remove('bg-[#0A2540]', 'text-white', 'shadow-sm');
        }
      });
      renderMapMarkers();
      updateLegend();
    }

    function updateLegend() {
      const cfg = LAYER_CONFIGS[currentLayer];
      const legendEl = document.getElementById('map-legend');
      legendEl.innerHTML = `

        <p class="text-[10px] text-slate-500 mb-2">${cfg.desc}</p>

              <span class="truncate">${item.label}</span>
            </div>
          `).join('')}
        </div>
      `;
    }

    function toggleDishMethod() {
      const method = document.querySelector('input[name="dish-method"]:checked').value;
      document.getElementById('dish-hand-box').classList.toggle('hidden', method !== 'hand');
      document.getElementById('dish-machine-box').classList.toggle('hidden', method !== 'machine');

    function calculateWater() {
      const showerMin = parseFloat(document.getElementById('in-shower-min').value) || 0;
      const showerFlow = parseFloat(document.getElementById('coef-shower-flow').value) || 9;
      const bathWeek = parseFloat(document.getElementById('in-bath-week').value) || 0;
      const handwashTimes = parseFloat(document.getElementById('in-handwash').value) || 0;
      const teethOpen = document.getElementById('in-teeth-open').checked;

      document.getElementById('val-shower-min').textContent = showerMin;

      const showerDaily = showerMin * showerFlow;
      const bathDaily = (bathWeek * 140) / 7;
      const handwashDaily = handwashTimes * 1.5;
      const teethDaily = teethOpen ? 12 : 1.5;
      const hygieneTotal = showerDaily + bathDaily + handwashDaily + teethDaily;

      const dishMethod = document.querySelector('input[name="dish-method"]:checked').value;
      const dishMin = parseFloat(document.getElementById('in-dish-min').value) || 0;
      const dishwasherWeek = parseFloat(document.getElementById('in-dishwasher-week').value) || 0;
      const vegLiters = parseFloat(document.getElementById('in-veg-liters').value) || 0;

      document.getElementById('val-dish-min').textContent = dishMin;
      document.getElementById('val-veg-liters').textContent = vegLiters;

      const dishDaily = dishMethod === 'hand' ? (dishMin * 6) : ((dishwasherWeek * 12) / 7);
      const kitchenTotal = dishDaily + vegLiters;

      const laundryWeek = parseFloat(document.getElementById('in-laundry-week').value) || 0;
      const laundryCoef = parseFloat(document.getElementById('coef-laundry-liters').value) || 50;
      const toiletDay = parseFloat(document.getElementById('in-toilet-day').value) || 0;
      const toiletCoef = parseFloat(document.getElementById('coef-toilet-liters').value) || 6;
      const cleaningLiters = parseFloat(document.getElementById('in-cleaning-liters').value) || 0;

      document.getElementById('val-cleaning-liters').textContent = cleaningLiters;

      const laundryDaily = (laundryWeek * laundryCoef) / 7;
      const toiletDaily = toiletDay * toiletCoef;
      const houseTotal = laundryDaily + toiletDaily + cleaningLiters;

      const dailyTotal = Math.round(hygieneTotal + kitchenTotal + houseTotal);
      const weeklyTotal = Math.round(dailyTotal * 7);
      const monthlyTotal = Math.round(dailyTotal * 30);

      document.getElementById('cat-hygiene-total').textContent = `${Math.round(hygieneTotal)} л/күн`;
      document.getElementById('cat-kitchen-total').textContent = `${Math.round(kitchenTotal)} л/күн`;
      document.getElementById('cat-house-total').textContent = `${Math.round(houseTotal)} л/күн`;

      document.getElementById('res-daily').textContent = dailyTotal.toLocaleString('ru-RU');
      document.getElementById('res-weekly').textContent = `${weeklyTotal.toLocaleString('ru-RU')} л`;
      document.getElementById('res-monthly').textContent = `${monthlyTotal.toLocaleString('ru-RU')} л`;

      const activities = [
        { name: 'Душ және ванна', val: showerDaily + bathDaily },
        { name: 'Дәретхана', val: toiletDaily },
        { name: 'Ыдыс жуу', val: kitchenTotal },
        { name: 'Кір жуу', val: laundryDaily },
        { name: 'Қол және тіс жуу', val: handwashDaily + teethDaily },
        { name: 'Үй жинау', val: cleaningLiters }
      ].sort((a, b) => b.val - a.val);

      document.getElementById('res-top-activity').textContent = `${activities[0].name} (~${Math.round(activities[0].val)} л)`;

      const tierBox = document.getElementById('res-tier-box');
      const tierTitle = document.getElementById('res-tier-title');
      const tierDesc = document.getElementById('res-tier-desc');

      if (dailyTotal <= 115) {
        tierBox.className = 'p-4 rounded-xl border border-emerald-200 bg-emerald-50 text-emerald-900';
        tierTitle.innerHTML = '🌿 «Сіз суды үнемді пайдаланасыз!»';
        tierDesc.textContent = 'Көрсеткішіңіз орташа деңгейден төмен және БҰҰ бағдарына жақын.';
      } else if (dailyTotal <= 165) {
        tierBox.className = 'p-4 rounded-xl border border-sky-200 bg-sky-50 text-sky-900';
        tierTitle.innerHTML = '💧 «Су тұтынуыңыз орташа деңгейде»';
        tierDesc.textContent = `Шығыныңыз қазақстандық орташа қалалық көрсеткішке (~145 л) сай келеді.`;
      } else {
        tierBox.className = 'p-4 rounded-xl border border-amber-200 bg-amber-50 text-amber-900';
        tierTitle.innerHTML = '⚠️ «Су тұтынуыңызды азайтуға болады»';
        tierDesc.textContent = 'Тұтыну орташа деңгейден жоғары. Душ уақытын қысқартуға кеңес беріледі.';
      }

      const chartData = [Math.round(hygieneTotal), Math.round(kitchenTotal), Math.round(houseTotal)];
      if (calcChart) {
        calcChart.data.datasets[0].data = chartData;
        calcChart.update();
      } else {
        const ctx = document.getElementById('calcDoughnutChart').getContext('2d');
        calcChart = new Chart(ctx, {
          type: 'doughnut',
          data: {
            labels: ['Жеке гигиена', 'Ас үй', 'Тұрмыстық'],
            datasets: [{
              data: chartData,
              backgroundColor: ['#0284C7', '#10B981', '#6366F1'],
              borderWidth: 2,
              borderColor: '#FFFFFF'
            }]
          },
          options: {
            responsive: true,
            maintainAspectRatio: false,
            cutout: '60%',
            plugins: { legend: { position: 'right', labels: { font: { family: 'Inter', size: 11 }, boxWidth: 12 } } }
          }
        });
      }
    }

    function updateEcoTracker() {
      const checkedBoxes = Array.from(document.querySelectorAll('.eco-task-cb:checked')).map(cb => cb.value);
      let points = 0;
      let liters = 0;

      checkedBoxes.forEach(id => {
        const task = ECO_TASKS.find(t => t.id === id);
        if (task) { points += task.points; liters += task.liters; }
      });

      document.getElementById('eco-points').textContent = points;
      document.getElementById('eco-liters-saved').textContent = `${liters} л`;
      document.getElementById('eco-percent-label').textContent = `${points}%`;
      document.getElementById('eco-progress-bar').style.width = `${points}%`;

      let rank = 'Бастаушы бақылаушы';
      if (points >= 80) rank = 'Эко-көшбасшы';
      else if (points >= 40) rank = 'Су сақшысы';
      else if (points > 0) rank = 'Жауапты тұтынушы';

      document.getElementById('eco-rank-label').textContent = `Мәртебе: ${rank}`;

      try { localStorage.setItem('aquakz_eco_tasks', JSON.stringify(checkedBoxes)); } catch (e) {}
    }

    function filterNews(cat) {
      document.querySelectorAll('.news-filter-btn').forEach(btn => {
        if (btn.dataset.filter === cat) {
          btn.classList.remove('bg-white', 'text-slate-700', 'border', 'border-slate-200');
          btn.classList.add('bg-[#0A2540]', 'text-white');
        } else {
          btn.classList.add('bg-white', 'text-slate-700', 'border', 'border-slate-200');
          btn.classList.remove('bg-[#0A2540]', 'text-white');
        }
      });
      renderNews(cat);
    }

    // 6. GLOBAL SEARCH
    function toggleMobileMenu() {
      const menu = document.getElementById('mobile-menu');
      menu.classList.toggle('hidden');
    }

    function openSearchModal() {
      const modal = document.getElementById('search-modal');
      modal.classList.remove('hidden');
      modal.classList.add('flex');
      const input = document.getElementById('global-search-input');
      input.value = '';
      input.focus();
      handleGlobalSearch('');
    }

    function closeSearchModal() {
      const modal = document.getElementById('search-modal');
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }

    function handleGlobalSearch(query) {
      const q = query.trim().toLowerCase();
      const resultsEl = document.getElementById('search-results-list');

      const cityMatches = CITIES_DATA.filter(c => c.name.toLowerCase().includes(q) || c.region.toLowerCase().includes(q)).slice(0, 5);
      const sectionLinks = [
        { title: 'Қазақстанның су картасы', href: '#map-section', sub: '20 қала мониторингі' },
        { title: 'Су тұтыну калькуляторы', href: '#calculator', sub: 'Тәуліктік литрді есептеу' },
        { title: 'Су статистикасы', href: '#statistics', sub: 'Ресми графиктер' },
        { title: 'Жоба туралы және Автор', href: '#about-project', sub: 'Ханабек Зере Қанатқызы (ҚазҰҚПУ)' }
      ].filter(s => !q || s.title.toLowerCase().includes(q));

      let html = '';
      if (cityMatches.length > 0) {
        html += `<p class="px-3 py-1 text-[10px] font-bold uppercase text-slate-400">Қалалар</p>`;
        html += cityMatches.map(c => `
          <button onclick="selectCityFromSearch('${c.id}')" class="w-full text-left px-3 py-2 rounded-xl hover:bg-sky-50 flex items-center justify-between">
            <span class="text-sm font-bold text-[#0A2540]">${c.name}</span>
            <span class="text-xs text-[#0284C7]">Картада көру →</span>
          </button>
        `).join('');
      }

      html += `<p class="px-3 py-1 mt-2 text-[10px] font-bold uppercase text-slate-400">Бөлімдер</p>`;
      html += sectionLinks.map(s => `
        <a href="${s.href}" onclick="closeSearchModal()" class="block px-3 py-2 rounded-xl hover:bg-slate-50">
          <p class="text-sm font-semibold text-slate-800">${s.title}</p>
        </a>
      `).join('');

      resultsEl.innerHTML = html;
    }

    function selectCityFromSearch(cityId) {
      closeSearchModal();
      document.getElementById('map-section').scrollIntoView({ behavior: 'smooth' });
      setTimeout(() => selectCityFromDropdown(cityId), 400);
    }

    document.addEventListener('keydown', (e) => {
      if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'k') {
        e.preventDefault();
        openSearchModal();
      }
      if (e.key === 'Escape') {
        closeSearchModal();
        closeTipModal();
      }
    });
